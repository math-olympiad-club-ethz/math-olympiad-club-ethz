\begin{problem}
Do there exist \(2024\) nonzero real numbers
\(a_1, a_2, \ldots, a_{2024}\), for which
\[
\sum_{i=1}^{2024} \left(a_i^2 + \frac{1}{a_i^2}\right)
\;+\;
2\sum_{i=1}^{2024} \frac{a_i}{a_{i+1}}
\;+\;
2024
\;=\;
2\sum_{i=1}^{2024} \left(a_i + \frac{1}{a_i}\right)\ ?
\]
(in the second sum \(a_{2025} = a_1\))
\end{problem}
