\begin{problem}
    Let \( f \in C^1\left( \left]a, b\right[, \mathbb{R} \right) \) with \( \lim_{x \to a^+}
    f\left(x\right) = +\infty \), \( \lim_{x \to b^-} f\left(x\right) = -\infty \), and
    \( f'\left(x\right) + f^2\left(x\right) \geq -1 \) for all \( x \in \left]a, b\right[ \). Prove
    that \( b - a \geq \pi \) and give an example where \( b - a = \pi \).
\end{problem}
\begin{solution}
    From the inequality, we obtain:
    \[
        \frac{\mathrm{d}}{\mathrm{d}x} \left( \arctan\left(f\left(x\right)\right) + x \right) =
        \frac{f'\left(x\right)}{1 + f^2\left(x\right)} + 1 \geq 0
    \]
    for all \( x \in \left]a, b\right[ \). Therefore, the function
    \( \arctan\left(f\left(x\right)\right) + x \) is increasing on \( \left]a, b\right[ \). Taking
    limits as \( x \) approaches the endpoints, we get:
    \[
        \lim_{x\underset{>}{\to} a} \left( \arctan\left(f\left(x\right)\right) + x \right) =
        \frac{\pi}{2} + a, \quad \lim_{x\underset{<}{\to} b} \left(
        \arctan\left(f\left(x\right)\right) + x \right) = -\frac{\pi}{2} + b.
    \]
    Hence,
    \[
        \frac{\pi}{2} + a \leq -\frac{\pi}{2} + b,
    \]
    which implies \( b - a \geq \pi \).
    \\\\
    Equality is achieved when:
    \[
        f\left(x\right) = \cot\left(x\right) = \frac{\cos\left(x\right)}{\sin\left(x\right)}, \quad
        a = 0, \quad b = \pi,
    \]
    since for any \( x \in \left]0, \pi\right[ \), we have:
    \[
        f'\left(x\right) + f^2\left(x\right) = -\frac{1}{\sin^2\left(x\right)} +
        \frac{\cos^2\left(x\right)}{\sin^2\left(x\right)} =
        -\frac{\sin^2\left(x\right)}{\sin^2\left(x\right)} = -1,
    \]
    and the boundary conditions are satisfied:
    \[
        \lim_{x \underset{>}{\to} 0^+} \cot\left(x\right) = +\infty, \quad
        \lim_{ x\underset{<}{\to} \pi} \cot\left(x\right) = -\infty.
    \]
\end{solution}
