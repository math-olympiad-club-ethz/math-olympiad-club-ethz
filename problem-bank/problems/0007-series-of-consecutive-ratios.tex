\begin{problem}
    Fix any \( a_0 \in \mathbb{R}_{>0} \) and define recursively, for all \( n \in
    \mathbb{N}_{\geq 1} \), assuming \( a_{n-1} \in \mathbb{R}_{>0} \) is defined,
    \[
        a_n := \frac{n a_{n-1}^2}{1 + \left(n+1\right) a_{n-1}},
    \]
    which clearly lies in \( \mathbb{R}_{>0} \). Show that
    \[
        \sum_{n=0}^{+\infty} \frac{a_{n+1}}{a_n}
    \]
    converges and find the limit.
\end{problem}
\begin{solution}
    First, we prove that \( \frac{a_{n+1}}{a_n} = \left(n+1\right) a_n - \left(n+2\right) a_{n+1} \)
    for all \( n \in \mathbb{N} \). By construction, for all \( n \in \mathbb{N} \),
    \[
        a_n^2 \left(n+1\right) = a_{n+1} \left(1 + \left(n+2\right) a_n\right) = a_{n+1} +
        \left(n+2\right) a_{n+1} a_n
    \]
    \[
        \implies a_{n+1} = \left(n+1\right) a_n^2 - \left(n+2\right) a_{n+1} a_n \implies
        \frac{a_{n+1}}{a_n} = \left(n+1\right) a_n - \left(n+2\right) a_{n+1}.
    \]
    Thus, for all \( N \in \mathbb{N} \),
    \[
        \sum_{n=0}^{N} \frac{a_{n+1}}{a_n} = \sum_{n=0}^{N} \left(\left(n+1\right) a_n -
        \left(n+2\right) a_{n+1}\right)
    \]
    \[
        = \left(\sum_{n=0}^{N} \left(n+1\right) a_n\right) - \left(\sum_{n=1}^{N+1} \left(n+1\right)
        a_n\right) = a_0 - \left(N+2\right) a_{N+1}.
    \]
    We now show that \( \left(n+1\right) a_n \overset{n \to +\infty}{\longrightarrow} 0 \) which
    implies that the partial sums converge to \( a_0 \). As \( \left(n+1\right) a_n -
    \left(n+2\right) a_{n+1} = \frac{a_{n+1}}{a_n} > 0 \) for all \( n \in \mathbb{N} \), we have
    \[
        \left(n+1\right) a_n > \left(n+2\right) a_{n+1} > 0,
    \]
    and hence \( \left\langle\left(n+1\right) a_n \mid n \in \mathbb{N}\right\rangle \) is a
    strictly decreasing bounded below sequence, and so classically, there exists \( L \in
    \mathbb{R}_{\geq 0} \) with \( L = \lim_{n \to +\infty} \left(n+1\right) a_n \). Since for all
    \( n \in \mathbb{N} \),
    \[
        \left(n+2\right) a_{n+1} =
        \frac{\left(n+2\right) \left(n+1\right) a_n^2}{1 + \left(n+2\right) a_n} =
        \frac{\frac{n+2}{n+1} \left(\left(n+1\right) a_n\right)^2}{1 + \frac{n+2}{n+1} \left(\left(n+1\right) a_n\right)},
    \]
    the limit \( L \) must satisfy (since \( 0<\frac{n+2}{n+1}
    \overset{n \to +\infty}{\longrightarrow} 1 \), \( 0<\left(n+1\right) a_n
    \overset{n \to +\infty}{\longrightarrow} L \) and \( \left(t, x\right) \mapsto
    \frac{t x^2}{1 + t x} \) is continuous on \( \mathbb{R}_{>0} \times \mathbb{R}_{\geq 0} \) (its
    denominator is at least \( 1 \)))
    \[
        L = \frac{L^2}{1+L}, \quad \text{i.e.} \quad L + L^2 = L^2, \quad \text{that is} \quad L =
        0.
    \]
\end{solution}
