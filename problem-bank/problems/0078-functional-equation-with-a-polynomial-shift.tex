\begin{problem}
    Let \( \mathbb{R}^+ \) be the set of all positive real numbers. Find all functions \( f :
    \mathbb{R}^+ \to \mathbb{R}^+ \) and polynomials \( P\left(x\right) \) with non-negative real
    coefficients such that \( P\left(0\right)=0 \) and which satisfy
    \[
        f\bigl(f\left(x\right) + P\left(y\right)\bigr) = f\left(x - y\right) + 2y
    \]
    for all real numbers \( x > y > 0 \).
\end{problem}
