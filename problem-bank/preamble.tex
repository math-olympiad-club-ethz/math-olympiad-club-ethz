% preamble.tex — shared LaTeX preamble of the Math Olympiad Club ETHZ problem bank.
%
% Used in two modes:
%   * AUTHOR / CI mode (default): a single problem file is wrapped by problems/_preview.tex
%     (or by build.py) and compiled with pdflatex + biber.  TikZ and biblatex are loaded so
%     inline figures and \cite work exactly as written.
%   * STITCHED mode (\BankStitched defined before \input{preamble}): the website assembles
%     many problems into one document and compiles it in the browser.  TikZ figures have been
%     externalised to PDF by CI and citations pre-resolved to plain text, so tikz/pgfplots and
%     biblatex are NOT loaded (they are the slow part of a compile).  This preamble is also
%     what the precompiled format file contains.
%
% Problem files contain NO preamble.  They use only the environments/macros defined here.

% ===== Deterministic PDF output (same input => byte-identical PDF) =====
% Measured with pdfTeX 1.40.29 (BusyTeX): \pdftrailerid{} (empty) still derives the /ID from the
% start time, so two compiles seconds apart differ in the trailer.  A non-empty argument fixes the
% /ID to the MD5 of that text.  The browser worker additionally sets SOURCE_DATE_EPOCH=0.
\ifdefined\pdfinfoomitdate
  \pdfinfoomitdate=1
  \pdftrailerid{Math Olympiad Club ETHZ problem bank}
  \pdfsuppressptexinfo=-1
\fi

% ===== Problem-bank environments =====
% Every problem file is:  header comments, \begin{problem}...\end{problem}, optional
% \begin{solution}...\end{solution}.  The wrapper (preview or stitched) prints the heading.
\newenvironment{problem}{\par\noindent\ignorespaces}{\par}
\newenvironment{solution}{\par\medskip\noindent\textbf{Solution.}\enspace\ignorespaces}{\par}
% Heading printed by the wrappers:  \bankproblem{<number in this PDF>}{<title>}{<origin, may be empty>}{<ID>}
% gives "Problem 3. Title" and, if known, the origin in grey below.  The ID (order of arrival in the repository)
% means nothing to readers: the website's PDFs (stitched mode) never print it; local previews and the CI proofs
% (author mode) show it as "Problem 3 (#0042). Title" so maintainers can find the file.
\ifdefined\BankStitched\newcommand{\bankidtag}[1]{}\else\newcommand{\bankidtag}[1]{ (\##1)}\fi
\newcommand{\bankproblem}[4]{%
  \par\addvspace{1.5\baselineskip}\penalty-300%
  \setcounter{equation}{0}\setcounter{theorem}{0}\setcounter{lemma}{0}\setcounter{definition}{0}%
  \setcounter{figure}{0}\setcounter{table}{0}\setcounter{footnote}{0}%
  \setcounter{section}{0}\setcounter{subsection}{0}\renewcommand{\thesection}{\arabic{section}}%
  \noindent{\Large\bfseries Problem #1\bankidtag{#4}.\ #2}\par\nopagebreak
  \ifx\relax#3\relax\else\noindent{\small\textcolor{gray}{#3}}\par\nopagebreak\fi
  \addvspace{0.5\baselineskip}}
% Optional line under the heading when the user asked to show the methods (spoilers, off by default).
\newcommand{\bankmethods}[1]{\noindent{\small\textcolor{gray}{Methods: #1}}\par\nopagebreak\addvspace{0.3\baselineskip}}
% "Not solved yet." marker used by the solutions PDF for unsolved / partial problems.
\newcommand{\banknosolution}{\par\medskip\noindent\textit{Not solved yet.}\par}
